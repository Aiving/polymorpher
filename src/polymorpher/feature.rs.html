1use std::fmt;
2
3use crate::{cubic::Cubic, geometry::PointTransformer};
4
5#[derive(Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord)]
6pub enum FeatureType {
7    Edge,
8    Corner { convex: bool },
9}
10
11/// While a polygon's shape can be drawn solely using a list of [`Cubic`]
12/// objects representing its raw curves and lines, features add an extra layer
13/// of context to groups of cubics.
14///
15/// Features group cubics into (straight) edges, convex corners, or concave
16/// corners. For example, rounding a rectangle adds many cubics around its
17/// edges, but the rectangle's overall number of corners remains the same.
18#[derive(Debug, Clone, PartialEq)]
19pub struct Feature {
20    pub ty: FeatureType,
21    pub cubics: Vec<Cubic>,
22}
23
24impl fmt::Display for Feature {
25    fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
26        match self.ty {
27            FeatureType::Edge => write!(f, "Edge"),
28            FeatureType::Corner { convex } => write!(
29                f,
30                "Corner: cubics={} convex={convex}",
31                self.cubics.iter().map(|c| format!("[{c}]")).collect::<Vec<_>>().join(", ")
32            ),
33        }
34    }
35}
36
37impl Feature {
38    pub const fn edge(cubics: Vec<Cubic>) -> Self {
39        Self { cubics, ty: FeatureType::Edge }
40    }
41
42    pub const fn corner(cubics: Vec<Cubic>, convex: bool) -> Self {
43        Self {
44            cubics,
45            ty: FeatureType::Corner { convex },
46        }
47    }
48
49    /// Returns a [`Feature`] with cubics transformed using the provided
50    /// reference to type that implements [`PointTransformer`] trait.
51    #[must_use]
52    pub fn transformed<T: PointTransformer>(self, f: &T) -> Self {
53        match self.ty {
54            FeatureType::Edge => Self {
55                cubics: self.cubics.into_iter().map(|cubic| cubic.transformed(f)).collect(),
56                ty: FeatureType::Edge,
57            },
58            FeatureType::Corner { convex } => Self {
59                cubics: self.cubics.into_iter().map(|cubic| cubic.transformed(f)).collect(),
60                ty: FeatureType::Corner { convex },
61            },
62        }
63    }
64
65    /// Returns `true` if the feature type is corner.
66    pub const fn is_corner(&self) -> bool {
67        matches!(self.ty, FeatureType::Corner { .. })
68    }
69
70    /// Returns the result of calling `func` if the type of feature is an angle
71    /// and passed to `func`, otherwise returning `false`.
72    pub fn is_corner_and<F: FnOnce(bool) -> bool>(&self, func: F) -> bool {
73        if let FeatureType::Corner { convex } = &self.ty {
74            func(*convex)
75        } else {
76            false
77        }
78    }
79}
